cmake_minimum_required(VERSION 3.21)
project(RaisingStudioQt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_library(studio-ui STATIC character-state-authoring.cpp roster-authoring.cpp studio.cpp export-dialog.cpp project-settings.cpp screen-settings.cpp user-fields.cpp text-binding-batch.cpp screens.cpp authoring.cpp progression-authoring.cpp sprite-import.cpp facilities.cpp rules.cpp logic-editor.cpp logic-workspace.cpp rule-debugger.cpp data-ui.cpp timeline-stage.cpp timeline.cpp battle-hud.cpp expedition-authoring.cpp script-authoring.cpp script-rounds.cpp validation.cpp hierarchy.cpp analysis.cpp route-tree.cpp design.cpp localization.cpp components.cpp renderer-preview.cpp canvas.cpp bridge.cpp plugin-host.cpp)
target_link_libraries(studio-ui PUBLIC Qt6::Widgets)
target_sources(studio-ui PRIVATE condition-preview.cpp)
target_include_directories(studio-ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(studio-ui PUBLIC /utf-8 /W4)
else()
  target_compile_options(studio-ui PRIVATE -Wall -Wextra)
endif()
add_executable(RaisingStudioQt WIN32 main.cpp)
target_link_libraries(RaisingStudioQt PRIVATE studio-ui)
add_executable(qt-studio-test tests.cpp)
target_link_libraries(qt-studio-test PRIVATE studio-ui Qt6::Test)
enable_testing()
add_test(NAME qt-studio COMMAND qt-studio-test)
