cmake_minimum_required(VERSION 3.20)
project(raising_addon LANGUAGES CXX)
add_subdirectory(.. core)
set_target_properties(raising_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(raising-addon MODULE addon.cpp)
target_include_directories(raising-addon PRIVATE vendor)
target_compile_features(raising-addon PRIVATE cxx_std_17)
target_compile_definitions(raising-addon PRIVATE NAPI_VERSION=8)
target_link_libraries(raising-addon PRIVATE raising_core)
set_target_properties(raising-addon PROPERTIES PREFIX "" SUFFIX ".node" OUTPUT_NAME "${RAISING_ADDON_NAME}")
if(MINGW)
  target_link_options(raising-addon PRIVATE -static -static-libgcc -static-libstdc++)
elseif(APPLE)
  target_link_options(raising-addon PRIVATE -undefined dynamic_lookup)
endif()
